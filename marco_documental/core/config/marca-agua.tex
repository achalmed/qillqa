\RequirePackage{eso-pic}

\newif\ifdespacho@marcaagua

\newcommand{\setmarcaagua}[1]{\renewcommand{\despacho@marcaagua@archivo}{#1}}

\AddToShipoutPictureBG{%

      }%
    }%
  \fi
}
