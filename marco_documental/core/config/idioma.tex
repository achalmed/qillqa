\RequirePackage{polyglossia}
\setdefaultlanguage[variant=spanish]{spanish}

% Comillas espanolas/inglesas coherentes
\RequirePackage[autostyle=false,style=american]{csquotes}

% --- Particion de palabras (guiones a fin de linea) ------------------
%  no : estilo Word/oficio, sin guiones (defecto)
%  si : particion normal de LaTeX
\newcommand{\setparticionpalabras}[1]{%
  \def\despacho@temp{#1}%
  \def\despacho@si{si}%
  \ifx\despacho@temp\despacho@si
    \hyphenpenalty=50\relax
    \tolerance=200\relax
    \emergencystretch=0pt\relax
  \else
    \hyphenpenalty=10000\relax
    \exhyphenpenalty=10000\relax
    \tolerance=1000\relax
    \emergencystretch=3em\relax
  \fi
}
\setparticionpalabras{no}
