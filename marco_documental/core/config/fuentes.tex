%% =====================================================================
%%  core/config/fuentes.tex  --  Tipografia propia del despacho
%% ---------------------------------------------------------------------
%%  Sistema tipografico de la marca (tokens/typography.json), adaptado
%%  al registro FORMAL de los documentos oficiales:
%%
%%    - Cuerpo:   Source Serif 4  (serif institucional; hermana de
%%                Source Sans 3, la fuente de cuerpo de la marca)
%%    - Display:  Archivo         (membrete, titulares, kickers)
%%    - Datos:    Inter           (cifras y tablas, numeros tabulares)
%%
%%  Las fuentes viajan VENDORIZADAS en assets/fonts/ y se cargan por
%%  NOMBRE DE ARCHIVO: scripts/build.sh exporta TTFONTS/OPENTYPEFONTS
%%  apuntando a la raiz del framework, asi el documento compila igual
%%  en cualquier maquina sin instalar nada.
%% =====================================================================

\RequirePackage{fontspec}

% --- Cuerpo: Source Serif 4 ------------------------------------------
\setmainfont{SourceSerif4-Regular.ttf}[
  BoldFont       = SourceSerif4-Bold.ttf,
  ItalicFont     = SourceSerif4-Italic.ttf,
  BoldItalicFont = SourceSerif4-BoldItalic.ttf,
  Ligatures      = TeX,
]

% Peso intermedio del cuerpo (destacados suaves, cargos)
\newfontfamily{\seriesemi}{SourceSerif4-SemiBold.ttf}[Ligatures=TeX]

% --- Display: Archivo (voz de marca) ---------------------------------
\newfontfamily{\fuentedisplay}{Archivo-Regular.ttf}[
  BoldFont   = Archivo-Bold.ttf,
  ItalicFont = Archivo-Italic.ttf,
  Ligatures  = TeX,
]
\newfontfamily{\fuentedisplaymedia}{Archivo-Medium.ttf}[Ligatures=TeX]
\newfontfamily{\fuentedisplaysemi}{Archivo-SemiBold.ttf}[Ligatures=TeX]
\newfontfamily{\fuentedisplaynegra}{Archivo-Black.ttf}[Ligatures=TeX]

% --- Membrete: Selawik (clon libre de Segoe UI, de Microsoft) --------
%%  Pedido de Edison (2026-08-03): nombre y cargo del membrete con
%%  tipografia "similar a Segoe UI Historic". Selawik es el reemplazo
%%  metrico oficial de Segoe UI publicado por Microsoft bajo SIL OFL.
\newfontfamily{\fuentemembrete}{Selawik-Regular.ttf}[
  BoldFont  = Selawik-Bold.ttf,
  Ligatures = TeX,
]
\newfontfamily{\fuentemembretesemi}{Selawik-SemiBold.ttf}[Ligatures=TeX]
\newfontfamily{\fuentemembretelig}{Selawik-SemiLight.ttf}[Ligatures=TeX]

% --- Datos: Inter con numeros tabulares (tablas, cifras) -------------
\newfontfamily{\fuentedatos}{Inter-Regular.ttf}[
  BoldFont   = Inter-Bold.ttf,
  Ligatures  = TeX,
  RawFeature = {+tnum},
]
\newfontfamily{\fuentedatossemi}{Inter-SemiBold.ttf}[
  Ligatures  = TeX,
  RawFeature = {+tnum},
]
