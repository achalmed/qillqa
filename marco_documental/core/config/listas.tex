\RequirePackage{enumitem}

%% La VINETA arranca exactamente en la sangria de parrafo (\parindent,
%% pedido de Edison 2026-08-04: antes leftmargin=2.2cm fijo entraba
%% demasiado); leftmargin=* se calcula solo (etiqueta + labelsep).
%% Ritmo ÚNICO (Edison 2026-10-08: «un interlineado único para el cuerpo,
%% entre párrafos y en listas»): el aire entre ítems y alrededor de la lista
%% es el mismo \parskip de los párrafos (fmt_espacioparrafos); antes era un
%% 2pt fijo que no seguía al ajuste. enumitem evalúa la clave al abrir cada
%% lista, así que toma el \parskip vigente. topsep=0pt: LaTeX ya pone el
%% \parskip antes y después de la lista; con topsep=\parskip se duplicaba.
\setlist{
  topsep=0pt,
  itemsep=\parskip,
  parsep=0pt,
  labelindent=\parindent,
  leftmargin=*,
  labelsep=0.5em,
}
\setlist[itemize,1]{label={\small\textcolor{tinta}{\rule[0.06em]{0.55em}{0.55em}}}}
\setlist[itemize,2]{label={\footnotesize$\circ$},
  labelindent=\dimexpr\parindent+1.5em\relax}
\setlist[enumerate,1]{label=\arabic*.}
%% Sub-lista ANIDADA: la etiqueta a) arranca AL RAS del texto del item
%% padre (pedido de Edison 2026-08-11: con labelindent=parindent+1.5em
%% el nivel 2 quedaba muy a la derecha, porque en enumitem la sangria
%% anidada se mide desde el margen del item padre, no desde la pagina).
\setlist[enumerate,2]{label=\alph*), labelindent=0pt}
