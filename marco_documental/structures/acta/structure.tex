%% ---------------------------------------------------------------------
%%  Estructura (norma de organos deliberantes peruanos):
%%
%%     ACTA DE SESIÓN N° ...            (titulo centrado, subrayado)
%%     Sesión Ordinaria · <organo>      (\organosesion)
%%     ... apertura + estaciones + acuerdos (cuerpo libre) ...
%%     \firmasacta{Pres}{cargo}{Sec}{cargo}   (doble firma lado a lado)
%%
%%  \estacion y \acuerdo dan formato a las estaciones y a los acuerdos
%%  correlativos. Guardia anti-doble-carga; nombres SIN @.
%% =====================================================================
\ifcsname despachoEstructuraActa\endcsname
  \expandafter\endinput
\fi
\expandafter\def\csname despachoEstructuraActa\endcsname{cargada}

%% --- Titulo del acta (centrado, subrayado) ---------------------------
%%  Usa \meta{numerocompleto} si esta; si no, compone tipodoc + N°.
\newcommand{\numerodocumento}{%
  \begin{center}
    {\seriesemi\underline{\MakeUppercase{%
      \ifmetadef{numerocompleto}{\meta{numerocompleto}}%
        {\meta{tipodoc}\ N°~\meta{numero}-\meta{anio}/\meta{siglas}}}}}%
  \end{center}
  \vspace{0.3\baselineskip}%
}

%% --- Clase de sesion + organo deliberante (centrado, cursiva) --------
\newcommand{\organosesion}[1]{%
  \begin{center}{\itshape #1}\end{center}
  \vspace{0.6\baselineskip}%
}

%% --- Estacion del acta (encabezado en mayusculas) --------------------
%%  p. ej. \estacion{Control de asistencia}
\newcommand{\estacion}[1]{%
  \vspace{0.7\baselineskip}%
  \noindent{\seriesemi\MakeUppercase{#1}}%
  \par\vspace{0.15\baselineskip}\noindent\ignorespaces
}

%% --- Acuerdo correlativo (en mayusculas y subrayado) -----------------
%%  \acuerdo{001-2026}{texto del acuerdo}
\newcommand{\acuerdo}[2]{%
  \vspace{0.4\baselineskip}%
  \noindent\underline{\textbf{ACUERDO N°~#1:}}~#2\par
}

%% --- Firmas del acta: presidente y secretario, lado a lado -----------
%%  \firmasacta{Nombre Presidente}{Cargo}{Nombre Secretario}{Cargo}
\newcommand{\firmasacta}[4]{%
  \par\vspace{1.8cm}%
  \noindent
  \begin{minipage}[t]{0.48\textwidth}\centering
    \rule{5.2cm}{0.4pt}\par
    {\seriesemi #1}\par #2\par
  \end{minipage}\hfill
  \begin{minipage}[t]{0.48\textwidth}\centering
    \rule{5.2cm}{0.4pt}\par
    {\seriesemi #3}\par #4\par
  \end{minipage}\par
}
